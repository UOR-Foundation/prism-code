structure{
  name: Project,
  type_parameters: [],

    {name: policies, type: Policies},

    {name: edges, type: (List Edge)},

  ]

inductive{
  name: Violation,
  type_parameters: [],
  parameters: [],
  constructors: [
    {name: DuplicateNode, fields: []},
    {name: UnknownNode, fields: []},
    {name: SelfEdge, fields: []},
    {name: DuplicateEdge, fields: []},
    {name: OutputSource, fields: []},
    {name: InputTarget, fields: []},
    {name: NotAnInput, fields: []},
    {name: DuplicatePii, fields: []},
    {name: DuplicatePolicy, fields: []}
  ]
}

inductive{
  name: Admission,
  type_parameters: [],
  parameters: [],
  constructors: [{name: Admit, fields: []}, {name: Refuse, fields: [Violation]}]
}

definition{
  name: admits,
  parameters: [admission : Admission],
  result: Bool,
  body: (match admission | Admission.Admit => true | Admission.Refuse violation => false)
}

definition{
  name: gate,
  parameters: [admission : Admission, candidate : Project],
  result: (Result Project Violation),
  body: (match admission
    | Admission.Admit => (ctor Result.ok [Project, Violation] candidate)
    | Admission.Refuse violation => (ctor Result.error [Project, Violation] violation))
}

definition{
  name: admitted,
  parameters: [outcome : (Result Project Violation)],
  result: Bool,
  body: match{
    branches: [
      {body: false, binders: [violation], constructor: Result.error},
      {body: true, binders: [project], constructor: Result.ok}
    ],
    scrutinee: outcome
  }
}

theorem{
  name: gateAdmitsExactly,
  parameters: [admission : Admission, candidate : Project],
  statement: ((admitted (gate admission candidate)) = (admits admission)),
  proof: cases{
    branches: [
      {proof: reflexivity{}, binders: [], constructor: Admit},
      {proof: reflexivity{}, binders: [violation], constructor: Refuse}
    ],
    scrutinee: admission
  }
}

definition{
  name: emptyProject,
  parameters: [],
  result: Project,
  body: (record Project {policies := (record Policies {memoryLimit := (ctor Option.none [UInt64]), piiEncryption := false}), nodes := (nil Node), edges := (nil Edge), pii := (nil String)})
}

definition{
  name: kindName,
  parameters: [kind : NodeKind],
  result: String,
  body: (match kind
    | NodeKind.Input => "input"
    | NodeKind.Operator => "operator"
    | NodeKind.Encrypt => "encrypt"
    | NodeKind.Output => "output")
}

definition{
  name: parseKind,
  parameters: [text : String],
  result: (Option NodeKind),
  body: (if (prim equal text "input" : Bool) then (ctor Option.some [NodeKind] (ctor NodeKind.Input)) else (if (prim equal text "operator" : Bool) then (ctor Option.some [NodeKind] (ctor NodeKind.Operator)) else (if (prim equal text "encrypt" : Bool) then (ctor Option.some [NodeKind] (ctor NodeKind.Encrypt)) else (if (prim equal text "output" : Bool) then (ctor Option.some [NodeKind] (ctor NodeKind.Output)) else (ctor Option.none [NodeKind])))))
}

definition{
  name: isInput,
  parameters: [kind : NodeKind],
  result: Bool,
  body: (match kind
    | NodeKind.Input => true
    | NodeKind.Operator => false
    | NodeKind.Encrypt => false
    | NodeKind.Output => false)
}

definition{
  name: isOperator,
  parameters: [kind : NodeKind],
  result: Bool,
  body: (match kind
    | NodeKind.Input => false
    | NodeKind.Operator => true
    | NodeKind.Encrypt => false
    | NodeKind.Output => false)
}

definition{
  name: isEncrypt,
  parameters: [kind : NodeKind],
  result: Bool,
  body: (match kind
    | NodeKind.Input => false
    | NodeKind.Operator => false
    | NodeKind.Encrypt => true
    | NodeKind.Output => false)
}

definition{
  name: isOutput,
  parameters: [kind : NodeKind],
  result: Bool,
  body: (match kind
    | NodeKind.Input => false
    | NodeKind.Operator => false
    | NodeKind.Encrypt => false
    | NodeKind.Output => true)
}

definition{
  name: findNode,
  parameters: [nodes : (List Node), name : String],
  result: (Option Node),
  body: (match nodes
    | List.nil => (ctor Option.none [Node])
    | List.cons head rest => (if (prim equal (project head name) name : Bool) then (ctor Option.some [Node] head) else (findNode rest name))),
  recursive_argument: nodes
}

definition{
  name: hasNode,
  parameters: [project : Project, name : String],
  result: Bool,
  body: (match (findNode (project var{name: project} nodes) name)
    | Option.none => false
    | Option.some node => true)
}

definition{
  name: listsName,
  parameters: [names : (List String), name : String],
  result: Bool,
  body: (match names
    | List.nil => false
    | List.cons head rest => (if (prim equal head name : Bool) then true else (listsName rest name))),
  recursive_argument: names
}

definition{
  name: hasEdge,
  parameters: [edges : (List Edge), source : String, target : String],
  result: Bool,
  body: (match edges
    | List.nil => false
    | List.cons head rest => (if ((prim equal (project head source) source : Bool) && (prim equal (project head target) target : Bool)) then true else (hasEdge rest source target))),
  recursive_argument: edges
}

definition{
  name: precedes,
  parameters: [left : String, right : String],
  result: Bool,
  body: (prim equal (prim compare_bytes (prim utf8_encode left : Bytes) (prim utf8_encode right : Bytes) : Ordering) (prim compare_bytes 0x 0x00 : Ordering) : Bool),
  axioms: [propext]
}

definition{
  name: edgePrecedes,
  parameters: [left : Edge, right : Edge],
  result: Bool,
  body: ((precedes (project left source) (project right source)) || ((prim equal (project left source) (project right source) : Bool) && (precedes (project left target) (project right target)))),
  axioms: [propext]
}

definition{
  name: insertNode,
  parameters: [item : Node, sorted : (List Node)],
  result: (List Node),
  body: (match sorted
    | List.nil => (cons item (nil Node))
    | List.cons head rest => (if (precedes (project item name) (project head name)) then (cons item (cons head rest)) else (cons head (insertNode item rest)))),
  axioms: [propext],
  recursive_argument: sorted
}

definition{
  name: sortNodes,
  parameters: [items : (List Node)],
  result: (List Node),
  body: (match items
    | List.nil => (nil Node)
    | List.cons head rest => (insertNode head (sortNodes rest))),
  axioms: [propext],
  recursive_argument: items
}

definition{
  name: insertEdge,
  parameters: [item : Edge, sorted : (List Edge)],
  result: (List Edge),
  body: (match sorted
    | List.nil => (cons item (nil Edge))
    | List.cons head rest => (if (edgePrecedes item head) then (cons item (cons head rest)) else (cons head (insertEdge item rest)))),
  axioms: [propext],
  recursive_argument: sorted
}

definition{
  name: sortEdges,
  parameters: [items : (List Edge)],
  result: (List Edge),
  body: (match items
    | List.nil => (nil Edge)
    | List.cons head rest => (insertEdge head (sortEdges rest))),
  axioms: [propext],
  recursive_argument: items
}

definition{
  name: insertName,
  parameters: [item : String, sorted : (List String)],
  result: (List String),
  body: (match sorted
    | List.nil => (cons item (nil String))
    | List.cons head rest => (if (precedes item head) then (cons item (cons head rest)) else (cons head (insertName item rest)))),
  axioms: [propext],
  recursive_argument: sorted
}

definition{
  name: sortNames,
  parameters: [items : (List String)],
  result: (List String),
  body: (match items
    | List.nil => (nil String)
    | List.cons head rest => (insertName head (sortNames rest))),
  axioms: [propext],
  recursive_argument: items
}

definition{
  name: nodeLine,
  parameters: [node : Node],
  result: String,
  body: (prim join (cons "node" (cons (project node name) (cons (kindName (project node kind)) (cons (prim format_decimal (project node memory) : String) (nil String))))) " " : String),
  axioms: [Classical.choice, Quot.sound, propext]
}

definition{
  name: edgeLine,
  parameters: [edge : Edge],
  result: String,
  body: (prim join (cons "edge" (cons (project edge source) (cons (project edge target) (nil String)))) " " : String),
  axioms: [propext]
}

definition{
  name: piiLine,
  parameters: [name : String],
  result: String,
  body: (prim join (cons "pii" (cons name (nil String))) " " : String),
  axioms: [propext]
}

definition{
  name: nodeLines,
  parameters: [items : (List Node)],
  result: (List String),
  body: (match items
    | List.nil => (nil String)
    | List.cons head rest => (cons (nodeLine head) (nodeLines rest))),
  axioms: [Classical.choice, Quot.sound, propext],
  recursive_argument: items
}

definition{
  name: edgeLines,
  parameters: [items : (List Edge)],
  result: (List String),
  body: (match items
    | List.nil => (nil String)
    | List.cons head rest => (cons (edgeLine head) (edgeLines rest))),
  axioms: [propext],
  recursive_argument: items
}

definition{
  name: piiLines,
  parameters: [items : (List String)],
  result: (List String),
  body: (match items
    | List.nil => (nil String)
    | List.cons head rest => (cons (piiLine head) (piiLines rest))),
  axioms: [propext],
  recursive_argument: items
}

definition{
  name: policyLines,
  parameters: [policies : Policies],
  result: (List String),
  body: (prim append (match (project policies memoryLimit)
    | Option.none => (nil String)
    | Option.some limit => (cons (prim join (cons "policy" (cons "memory" (cons (prim format_decimal limit : String) (nil String)))) " " : String) (nil String))) (if (project policies piiEncryption) then (cons "policy encrypt-pii" (nil String)) else (nil String)) : (List String)),
  axioms: [Classical.choice, Quot.sound, propext]
}

definition{
  name: canonicalLines,
  parameters: [project : Project],
  result: (List String),
  body: (cons "dataflow-policy/1" (prim append (policyLines (project var{
    name: project
  } policies)) (prim append (nodeLines (sortNodes (project var{
    name: project
  } nodes))) (prim append (piiLines (sortNames (project var{
    name: project
  } pii))) (edgeLines (sortEdges (project var{
    name: project
  } edges))) : (List String)) : (List String)) : (List String))),
  axioms: [Classical.choice, Quot.sound, propext]
}
